# API security: OAuth 2.0 bearer tokens (JWT). The API is a resource server --
# it validates signed access tokens and their scopes. Swapping the local token
# issuer for a hosted IdP means verifying against that provider's JWKS instead
# of a local secret; nothing else changes.
pyjwt==2.13.0

# Database. PostgreSQL is the live application database in both development
# and production (see backend/storage.py) -- there is no SQLite fallback.
# psycopg2-binary bundles libpq so nothing extra needs installing to connect;
# the plain (non-binary) package is recommended for production builds that
# compile their own dependencies, but this project has no such build step.
psycopg2-binary==2.9.10

# Email security (backend/email_security.py). NOT a new dependency: pdfplumber
# already pulls this in through pdfminer.six, so every working install of this
# project has had it all along. Declared explicitly because Phase F now USES it
# directly -- for real RFC 6376 DKIM signature verification -- and a direct use
# should not rest on another package's transitive choice.
cryptography==50.0.0

# Live DNS for DKIM public keys and DMARC policies (EMAIL_DNS_RESOLVER=dnspython).
# NOT installed by default, exactly like boto3 above: the default resolver does
# no lookups at all and reports DKIM `unavailable` rather than failing, so an
# install that does not opt in never needs the package. DnspythonTxtResolver
# imports it lazily and raises a clear error at construction if it is missing.
# dnspython==2.7.0

# Document storage (backend/documents.py). NOT installed by default: the
# local filesystem backend (DOCUMENT_STORE_BACKEND=local, the default) needs
# nothing beyond the standard library. Uncomment only when running with
# DOCUMENT_STORE_BACKEND=s3, which imports boto3 lazily and raises a clear
# error at call time if it is missing rather than failing every import.
